cmake_minimum_required(VERSION 3.24)
project(GRIDAutosportVR LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use MSVC with -A Win32")
endif()
set(OPENXR_SDK "" CACHE PATH "Khronos OpenXR-SDK checkout")
if(NOT EXISTS "${OPENXR_SDK}/include/openxr/openxr.h")
  message(FATAL_ERROR "Set OPENXR_SDK to the Khronos source checkout")
endif()
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
set(DYNAMIC_LOADER OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENXR_SDK}" openxr-sdk EXCLUDE_FROM_ALL)
add_library(gasxr SHARED game_device.cpp native_ovr.cpp loading_overlay.cpp device_claim.cpp cockpit_textures.cpp opponent_lod.cpp seat.cpp xr_bridge.cpp eye_dump.cpp gasxr.def)
target_compile_features(gasxr PRIVATE cxx_std_17)
target_compile_definitions(gasxr PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
target_compile_options(gasxr PRIVATE /W4 /EHsc)
target_link_libraries(gasxr PRIVATE openxr_loader d3d11 dxgi d3dcompiler)
